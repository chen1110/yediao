《夜钓》搭建说明

这是一款完整的竖屏小游戏。请完全按本页实现：直接使用第 3 节的完整代码，不要改写、不要优化、不要替换任何数值、颜色和文字。代码已经可以直接运行，在线试玩：https://chen1110.github.io/yediao/play.html

1. 规格

  • 设计画布 720×1570 竖屏，整体等比缩放居中，四周用 #0E1630 填充。
  • 右上角 200×110 是平台按钮区，不放任何内容。
  • 玩法：点击屏幕抛竿 → 等鱼咬钩 → 出现张力条，指针在色块内时点击即可收线。每局 60 秒，结束显示结算。

2. 素材（3 张，按原图使用，不重画）

3. 完整代码（单个 HTML 文件，原样使用）

<!doctype html>
<html lang="zh-CN"><head><meta charset="utf-8">
<meta name="viewport" content="width=device-width,initial-scale=1,maximum-scale=1,user-scalable=no">

<script>
// 《夜钓》v0 —— 单文件 Canvas 小游戏。设计基准 720x1570 竖屏，等比缩放。
// 素材：bg.jpg(720x1570 背景) / fish.png(720x320，3列x2行，每格240x160) / ui.png(640x531 组件合图，坐标见 UI)
const CFG = {"version":"A-网页版","subtitle":"月亮在看，鱼也在看","emptyText":"第一竿总是空的","eggCode":"夜钓 YD-A-0427","barColor":"#7B2FBE","barHz":1.37,"zoneW":0.23,"roundSec":60,"rodTip":[620,992],"assets":{"bg":"assets/bg.jpg","fish":"assets/fish.png","ui":"assets/ui.png"},"fish":[{"id":0,"name":"小白条","p":35,"kg":[0.1,0.3],"coin":1,"line":"它看起来比你还困。"},{"id":1,"name":"斑纹鲈","p":25,"kg":[0.4,1.2],"coin":3,"line":"条纹很整齐，像排班表。"},{"id":2,"name":"镜面鲈","p":15,"kg":[0.5,1.5],"coin":5,"line":"你在它身上看到了自己的黑眼圈。"},{"id":3,"name":"鼓气鲀","p":12,"kg":[0.3,0.9],"coin":6,"line":"它很生气，但也只能生气。"},{"id":4,"name":"夜行鲶","p":9,"kg":[1.5,4.0],"coin":10,"line":"自带照明，省了你的手电。"},{"id":5,"name":"金鳞锦","p":4,"kg":[2.0,5.0],"coin":30,"line":"今晚的运气，全在这一条了。"}]};

const W = 720, H = 1570;
const UI = { // ui.png 里每个组件的位置 [x,y,w,h]
  gauge: [0, 0, 640, 100], panel: [0, 100, 600, 296], button: [0, 396, 360, 135],
  bobber: [360, 396, 60, 96], bang: [420, 396, 80, 90], badge: [500, 396, 100, 100]
};
const ROD_TIP = [CFG.rodTip[0], CFG.rodTip[1]];
const COLORS = { ink: '#0E1630', paper: '#F6F3EA', lantern: '#F2A541', white: '#FFFFFF', red: '#D64545' };

const cv = document.getElementById('game');
const ctx = cv.getContext('2d');
const img = {};
let scale = 1, offX = 0, offY = 0;

function load(name, src) {
  return new Promise(r => { const i = new Image(); i.onload = () => { img[name] = i; r(); }; i.onerror = () => r(); i.src = src; });
}
function resize() {
  const dpr = window.devicePixelRatio || 1;
  cv.width = innerWidth * dpr; cv.height = innerHeight * dpr;
  cv.style.width = innerWidth + 'px'; cv.style.height = innerHeight + 'px';
  scale = Math.min(innerWidth / W, innerHeight / H);
  offX = (innerWidth - W * scale) / 2; offY = (innerHeight - H * scale) / 2;
  ctx.setTransform(dpr * scale, 0, 0, dpr * scale, dpr * offX, dpr * offY);
}

// ---------- 状态 ----------
let S = 'title';            // title | play | result
let phase = 'ready';        // ready | fly | wait | reel
let t0 = 0, timeLeft = CFG.roundSec, castNo = 0;
let bob = { x: 360, y: 820 }, flyFrom = 0, biteAt = 0, reelStart = 0, zoneA = 0.4;
let catches = [], popup = null, toast = null, eggTaps = [];

function now() { return performance.now() / 1000; }
function rand(a, b) { return a + Math.random() * (b - a); }
function say(text, sec = 1.4) { toast = { text, until: now() + sec }; }

function startRound() {
  S = 'play'; phase = 'ready'; t0 = now(); timeLeft = CFG.roundSec; castNo = 0; catches = []; popup = null; toast = null;
}
function cast() {
  phase = 'fly'; flyFrom = now();
  bob = { x: rand(180, 540), y: rand(580, 800) };
}
function rollFish() {
  const total = CFG.fish.reduce((s, f) => s + f.p, 0);
  let r = Math.random() * total;
  for (const f of CFG.fish) { if ((r -= f.p) < 0) return f; }
  return CFG.fish[0];
}
function needle() { // 0..1 来回摆动，每秒 CFG.barHz 个来回
  return (1 - Math.cos(2 * Math.PI * CFG.barHz * (now() - reelStart))) / 2;
}
function endRound() { S = 'result'; phase = 'ready'; popup = null; }

// ---------- 输入 ----------
function tap(px, py) {
  const x = (px - offX) / scale, y = (py - offY) / scale;
  if (S === 'title') {
    if (x < 160 && y < 260) { // 彩蛋：左上角 1.5 秒内连点 5 下
      const t = now(); eggTaps = eggTaps.filter(v => t - v < 1.5); eggTaps.push(t);
      if (eggTaps.length >= 5) { eggTaps = []; say(CFG.eggCode, 3); }
      return;
    }
    if (inBtn(x, y, 900)) startRound();
    return;
  }
  if (S === 'result') { if (inBtn(x, y, 1010)) startRound(); return; }
  if (popup) { popup = null; return; }
  if (phase === 'ready') cast();
  else if (phase === 'wait') { phase = 'ready'; say('太急了，鱼被吓跑了'); }
  else if (phase === 'reel') {
    const p = needle();
    if (p >= zoneA && p <= zoneA + CFG.zoneW) {
      const f = rollFish();
      const kg = Math.round(rand(f.kg[0], f.kg[1]) * 100) / 100;
      catches.push({ f, kg }); popup = { f, kg, until: now() + 1.8 };
    } else say('线松了，鱼跑了');
    phase = 'ready';
  }
}
function inBtn(x, y, cy) { return x > 180 && x < 540 && y > cy - 60 && y < cy + 60; }
cv.addEventListener('pointerdown', e => { e.preventDefault(); tap(e.clientX, e.clientY); });

// ---------- 逻辑 ----------
function update() {
  const t = now();
  if (S !== 'play') return;
  timeLeft = Math.max(0, CFG.roundSec - (t - t0));
  if (timeLeft <= 0) return endRound();
  if (popup && t > popup.until) popup = null;
  if (phase === 'fly' && t - flyFrom > 0.4) {
    phase = 'wait';
    if (castNo === 0) biteAt = -1; else biteAt = t + rand(1, 3);
    castNo++;
    if (biteAt < 0) setTimeout(() => { if (S === 'play' && phase === 'wait') { phase = 'ready'; say(CFG.emptyText, 2); } }, 2200);
  }
  if (phase === 'wait' && biteAt > 0 && t >= biteAt) {
    phase = 'reel'; reelStart = t; zoneA = rand(0.08, 0.92 - CFG.zoneW);
  }
  if (phase === 'reel' && t - reelStart > 3) { phase = 'ready'; say('鱼挣脱了'); }
}

// ---------- 绘制 ----------
function sprite(name, r, dx, dy, dw, dh) { if (img[name]) ctx.drawImage(img[name], r[0], r[1], r[2], r[3], dx, dy, dw, dh); }
function fishRect(i) { return [(i % 3) * 240, Math.floor(i / 3) * 160, 240, 160]; }
function text(s, x, y, size, color, align = 'center', weight = 'bold', stroke = true) {
  ctx.font = `${weight} ${size}px "PingFang SC","Microsoft YaHei",sans-serif`;
  ctx.textAlign = align; ctx.textBaseline = 'middle';
  if (stroke) { ctx.lineWidth = Math.max(3, size / 8); ctx.strokeStyle = 'rgba(14,22,48,0.85)'; ctx.strokeText(s, x, y); }
  ctx.fillStyle = color; ctx.fillText(s, x, y);
}
function button(label, cy) {
  sprite('ui', UI.button, 180, cy - 67, 360, 135);
  text(label, 360, cy - 8, 40, COLORS.ink, 'center', 'bold', false);
}
function drawLine(x, y) {
  ctx.strokeStyle = 'rgba(246,243,234,0.7)'; ctx.lineWidth = 2;
  ctx.beginPath(); ctx.moveTo(ROD_TIP[0], ROD_TIP[1]);
  ctx.quadraticCurveTo((ROD_TIP[0] + x) / 2, Math.min(ROD_TIP[1], y) - 60, x, y); ctx.stroke();
}
function render() {
  const t = now();
  ctx.fillStyle = COLORS.ink; ctx.fillRect(-2000, -2000, W + 4000, H + 4000);
  if (img.bg) ctx.drawImage(img.bg, 0, 0, W, H);

  if (S === 'title') {
    text('夜 钓', 360, 600, 120, COLORS.paper);
    text(CFG.subtitle, 360, 700, 30, COLORS.lantern, 'center', 'normal');
    button('开始夜钓', 900);
    text('60 秒，能钓几条是几条', 360, 1000, 26, COLORS.paper, 'center', 'normal');
  }

  if (S === 'play') {
    // 顶部左侧 HUD（右上角 200x110 是平台胶囊，不放东西）
    text(`⏱ ${Math.ceil(timeLeft)}s`, 32, 150, 40, timeLeft <= 10 ? COLORS.red : COLORS.paper, 'left');
    const kg = catches.reduce((s, c) => s + c.kg, 0);
    text(`渔获 ${catches.length} 条 · ${kg.toFixed(2)} kg`, 32, 205, 26, COLORS.paper, 'left', 'normal');

    if (phase === 'fly') {
      const k = Math.min(1, (t - flyFrom) / 0.4);
      const x = ROD_TIP[0] + (bob.x - ROD_TIP[0]) * k, y = ROD_TIP[1] + (bob.y - ROD_TIP[1]) * k - Math.sin(k * Math.PI) * 220;
      drawLine(x, y); sprite('ui', UI.bobber, x - 20, y - 32, 40, 64);
    }
    if (phase === 'wait' || phase === 'reel') {
      const dip = phase === 'reel' ? 14 : Math.sin(t * 3) * 4;
      drawLine(bob.x, bob.y + dip);
      sprite('ui', UI.bobber, bob.x - 20, bob.y - 32 + dip, 40, 64);
      ctx.strokeStyle = 'rgba(246,243,234,0.35)'; ctx.lineWidth = 2;
      ctx.beginPath(); ctx.ellipse(bob.x, bob.y + 32, 30 + (t * 20) % 20, 8, 0, 0, Math.PI * 2); ctx.stroke();
    }
    if (phase === 'reel') {
      sprite('ui', UI.bang, bob.x + 16, bob.y - 104, 60, 68);
      // 张力条：先画槽和色块，再盖外框贴图（外框中间透明），最后画指针
      const gx = 66, gy = 878, gw = 588, gh = 59;
      ctx.fillStyle = 'rgba(14,22,48,0.9)'; ctx.fillRect(gx, gy, gw, gh);
      ctx.fillStyle = CFG.barColor; ctx.fillRect(gx + gw * zoneA, gy, gw * CFG.zoneW, gh);
      sprite('ui', UI.gauge, 40, 860, 640, 100);
      const nx = gx + gw * needle();
      ctx.fillStyle = COLORS.white; ctx.fillRect(nx - 4, gy - 12, 8, gh + 24);
      text('在色块里点一下收线！', 360, 1000, 30, COLORS.paper);
    }
    if (phase === 'ready' && !popup) text('点击屏幕抛竿', 360, 960, 36, COLORS.paper);

    if (popup) {
      sprite('ui', UI.panel, 60, 500, 600, 336);
      sprite('fish', fishRect(popup.f.id), 240, 520, 240, 160);
      text(`${popup.f.name}  ${popup.kg.toFixed(2)} kg`, 360, 712, 40, COLORS.ink, 'center', 'bold', false);
      text(popup.f.line, 360, 772, 26, COLORS.ink, 'center', 'normal', false);
    }
  }

  if (S === 'result') {
    sprite('ui', UI.panel, 40, 360, 640, 540);
    text('今晚收竿', 360, 450, 64, COLORS.ink, 'center', 'bold', false);
    const kg = catches.reduce((s, c) => s + c.kg, 0);
    text(`${catches.length} 条 · ${kg.toFixed(2)} kg`, 360, 530, 40, COLORS.ink, 'center', 'bold', false);
    const best = catches.slice().sort((a, b) => b.f.id - a.f.id || b.kg - a.kg)[0];
    if (best) {
      sprite('fish', fishRect(best.f.id), 240, 570, 240, 160);
      text(`最佳：${best.f.name} ${best.kg.toFixed(2)} kg`, 360, 760, 30, COLORS.ink, 'center', 'normal', false);
    } else text('空军了。月亮作证。', 360, 660, 30, COLORS.ink, 'center', 'normal', false);
    const coin = catches.reduce((s, c) => s + c.f.coin, 0);
    text(`获得金币 ${coin}`, 360, 830, 36, '#B07A1E', 'center', 'bold', false);
    button('再来一局', 1010);
  }

  if (toast && t < toast.until) text(toast.text, 360, 700, 34, COLORS.lantern);
}

function loop() { update(); render(); requestAnimationFrame(loop); }
addEventListener('resize', resize);
resize();
Promise.all([load('bg', CFG.assets.bg), load('fish', CFG.assets.fish), load('ui', CFG.assets.ui)]).then(loop);

</script></body></html>
